import authenticate from './middleware/authenticate';
import { createDocumentsRouter } from './modules/documents/documents.routes';
import { createDocumentUpload } from './modules/documents/upload.middleware';
import { DocumentService } from './modules/documents/document.service';
import { createGoogleDriveStorage } from './modules/documents/drive.service';
import { PrismaAttachmentRepository } from './modules/documents/prisma-attachment.repository';

const documentService = new DocumentService({
  drive: createGoogleDriveStorage(),            // needs GOOGLE_DRIVE_FOLDER_ID + GOOGLE_APPLICATION_CREDENTIALS
  repository: new PrismaAttachmentRepository(),
});

app.use('/api', createDocumentsRouter({
  service: documentService,
  access: contractDocumentAccess,               // from Member 1: who may view / upload for a contract
  authenticate,
  upload: createDocumentUpload(),               // throws at start-up if DOCUMENT_* settings are missing
}));

  GOOGLE_DRIVE_FOLDER_ID, GOOGLE_APPLICATION_CREDENTIALS
